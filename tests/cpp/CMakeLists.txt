cmake_minimum_required(VERSION 3.26)
project(ffi_buffer_ownership LANGUAGES CXX)
find_package(tvm_ffi CONFIG REQUIRED)
add_executable(ffi_list_buffer_ownership ffi_list_buffer_ownership.cc)
target_compile_features(ffi_list_buffer_ownership PRIVATE cxx_std_20)
target_link_libraries(ffi_list_buffer_ownership PRIVATE tvm_ffi::shared)
option(TVM_CHECK_FUNCTOR_OWNERSHIP "Check the split core's shared dispatch table" OFF)
if(TVM_CHECK_FUNCTOR_OWNERSHIP)
  find_package(tvm_tirx CONFIG REQUIRED)
  add_executable(object_functor_buffer_ownership object_functor_buffer_ownership.cc)
  target_compile_features(object_functor_buffer_ownership PRIVATE cxx_std_20)
  target_link_libraries(object_functor_buffer_ownership PRIVATE tvm_tirx_compiler::shared tvm_ffi::shared)
endif()
